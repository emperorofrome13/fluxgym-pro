1: """
2: fluxgym-pro command generator.
3: 
4: Pure-Python, no torch / no GPU: builds the exact training command sequence for
5: each architecture.
6: 
7: Backends (verified against upstream docs, 2026-02):
8:   - Z-Image        -> musubi-tuner  (zimage_cache_latents.py, zimage_train_network.py)
9:   - Krea 2         -> musubi-tuner  (krea2_cache_latents.py,   krea2_train_network.py)
10:   - Qwen-Image-2.1 -> ai-toolkit    (run.py + generated YAML; musubi-tuner has no
11:                                       Qwen-Image-2.1 support yet, only the older
12:                                       20B Qwen-Image)
13: """
14: 
15: import os
16: 
17: # arch -> musubi-tuner script stems. qwen_image_2 is handled by ai-toolkit.
18: MUSUBI_SCRIPTS = {
19:     "zimage": {
20:         "cache_latents": "zimage_cache_latents.py",
21:         "cache_text": "zimage_cache_text_encoder_outputs.py",
22:         "train": "zimage_train_network.py",
23:     },
24:     "krea2": {
25:         "cache_latents": "krea2_cache_latents.py",
26:         "cache_text": "krea2_cache_text_encoder_outputs.py",
27:         "train": "krea2_train_network.py",
28:     },
29: }
30: 
31: # arch -> network module inside musubi-tuner's networks/ package
32: NETWORK_MODULE = {"zimage": "networks.lora_zimage", "krea2": "networks.lora_krea2"}
33: 
34: # flow-matching sampling settings per musubi arch (straight from upstream docs)
35: FLOW_ARGS = {
36:     "zimage": "--timestep_sampling shift --weighting_scheme none --discrete_flow_shift 2.0",
37:     "krea2": "--timestep_sampling shift --weighting_scheme none --discrete_flow_shift 2.5",
38: }
39: 
40: ARCH_LABELS = {
41:     "zimage": "Z-Image",
42:     "krea2": "Krea 2",
43:     "qwen_image_2": "Qwen-Image-2.1",
44: }

46: 
47: def _ps(lines):
48:     """Join command lines with PowerShell continuation backticks (no dangling)."""
49:     return " `\n  ".join(lines)
50: 
51: 
52: def _ps_quote(s):
53:     """Quote a value as a single PowerShell argument.
54: 
55:     Doubles backticks and escapes double quotes so a value from the UI can
56:     never terminate the quoted argument or escape into code. CR/LF are
57:     neutralized: inside a double-quoted string a raw newline would break the
58:     argument (and the line) apart.
59:     """
60:     return ('"' + str(s).replace("\r\n", " ").replace("\r", " ")
61:             .replace("\n", " ").replace("`", "``").replace('"', '`"') + '"')
62: 
63: 
64: def _yaml_str(s):
65:     """Escape a value for a YAML double-quoted scalar (no newlines, no breaks)."""
66:     return (str(s).replace("\r", " ").replace("\n", " ")
67:             .replace("\\", "\\\\").replace('"', '\\"'))
68: 
69: 
70: def dataset_toml(train_dir, resolution, batch_size, repeats=1):
71:     """musubi-tuner dataset config (docs/dataset_config.md schema)."""
72:     r = f"[{resolution}, {resolution}]"
73:     return (
74:         "# generated by fluxgym-pro\n"
75:         "[general]\n"
76:         f"resolution = {r}\n"
77:         "caption_extension = \".txt\"\n"
78:         f"batch_size = {batch_size}\n"
79:         "\n"
80:         "[[datasets]]\n"
81:         f"image_directory = \"{os.path.abspath(os.path.join(train_dir, 'img')).replace(os.sep, '/')}\"\n"
82:         f"num_repeats = {repeats}\n"
83:     )
84:

86: def _musubi_commands(cfg):
87:     """musubi-tuner path: cache latents -> cache text encoder outputs -> train.
88: 
89:     Scripts run from the musubi-tuner repo root as `python src/musubi_tuner/<script>`.
90:     """
91:     arch = cfg["arch"]
92:     scripts = MUSUBI_SCRIPTS[arch]
93:     toml = cfg["toml_path"]
94:     py = cfg["python_exe"]
95:     out_dir = cfg["output_dir"]
96:     out_name = cfg["output_name"]
97:     epochs = cfg["epochs"]
98:     lr = cfg["learning_rate"]
99:     dim = cfg["network_dim"]
100:     alpha = int(cfg.get("network_alpha") or dim)
101:     seed = cfg["seed"]
102: 
103:     vae = cfg.get("vae_path")
104:     te = cfg.get("text_encoder_path")
105:     dit = cfg.get("dit_path")
106:     save_every = cfg.get("save_every_n_epochs", 1)
107:     prompts = cfg.get("sample_prompts") or []
108:     sample_every = int(cfg.get("sample_every") or 0)
109:     sampling = bool(prompts) and sample_every > 0
110:     prompts_path = ""
111:     if sampling:
112:         # command_gen may be called directly (tests, previews); keep its own
113:         # file writes safe even if the caller skipped build_cfg validation.
114:         fs_name = "".join(c if c not in '\\/:*?"<>|' else "_" for c in out_name)
115:         prompts_path = os.path.join(cfg["output_dir"],
116:                                     f"{fs_name}_sample_prompts.txt")
117:         with open(prompts_path, "w", encoding="utf-8") as f:
118:             for p in prompts:
119:                 f.write(p.replace("\n", " ").strip() + "\n")
120: 
121:     if not (vae and os.path.exists(vae)):
122:         raise RuntimeError(f"VAE weights not found: {vae}")
123:     if not (te and os.path.exists(te)):
124:         raise RuntimeError(f"Text encoder weights not found: {te}")
125:     if not (dit and os.path.exists(dit)):
126:         raise RuntimeError(f"DiT weights not found: {dit}")
127: 
128:     cache_lat = _ps([f"{py}", "src/musubi_tuner/" + scripts["cache_latents"],
129:                      "--dataset_config", _ps_quote(toml), "--vae", _ps_quote(vae)])
130:     cache_te = _ps([f"{py}", "src/musubi_tuner/" + scripts["cache_text"],
131:                     "--dataset_config", _ps_quote(toml), "--text_encoder", _ps_quote(te),
132:                     "--batch_size", "16"])
133:     train = _ps([
134:         "accelerate launch --num_cpu_threads_per_process 1 --mixed_precision bf16",
135:         "src/musubi_tuner/" + scripts["train"],
136:         "--dit", _ps_quote(dit),
137:         "--vae", _ps_quote(vae),
138:         # krea2 only needs the TE during training if sample images are generated
139:         *(["--text_encoder", _ps_quote(te)] if (arch != "krea2" or sampling) else []),
140:         "--dataset_config", _ps_quote(toml),
141:         "--sdpa --mixed_precision bf16",
142:         *(["--fp8_base --fp8_scaled"] if cfg.get("fp8_base") else []),
143:         FLOW_ARGS[arch],
144:         "--optimizer_type adamw8bit",
145:         "--learning_rate", str(lr),
146:         "--gradient_checkpointing",
147:         "--max_data_loader_n_workers 2 --persistent_data_loader_workers",
148:         "--network_module", NETWORK_MODULE[arch],
149:         "--network_dim", str(dim),
150:         "--network_alpha", str(alpha),
151:         "--max_train_epochs", str(epochs),
152:         f"--save_every_n_epochs {save_every} --save_state",
153:         *(["--sample_prompts", _ps_quote(prompts_path), "--sample_every_n_epochs", str(sample_every)]
154:           if sampling else []),
155:         "--seed", str(seed),
156:         "--output_dir", _ps_quote(out_dir),
157:         "--output_name", _ps_quote(out_name),
158:     ])
159: 
160:     body = (
161:         f'Write-Host "=== [1/3] caching latents ({ARCH_LABELS[arch]}) ==="\n'
162:         f"{cache_lat}\n"
163:         f'if ($LASTEXITCODE -ne 0) {{ throw "latent caching failed" }}\n\n'
164:         f'Write-Host "=== [2/3] caching text encoder outputs ==="\n'
165:         f"{cache_te}\n"
166:         f'if ($LASTEXITCODE -ne 0) {{ throw "text encoder caching failed" }}\n\n'
167:         f'Write-Host "=== [3/3] training LoRA ==="\n'
168:         f"{train}\n"
169:         f'if ($LASTEXITCODE -ne 0) {{ throw "training failed" }}\n\n'
170:         f'Write-Host "DONE: {out_dir}\\{out_name}.safetensors"\n'
171:     )
172:     return body, "musubi-tuner"

175: def _atk_yaml(cfg):
176:     """ai-toolkit training config for Qwen-Image-2.1 (day-0 upstream support).
177: 
178:     Schema mirrors ai-toolkit's sd_trainer examples: job=extension, one
179:     sd_trainer process, model.arch selects the architecture implementation.
180:     """
181:     arch = "qwen_image_2"
182:     base_repo = cfg["repo"]
183:     train_dir = cfg["train_data_dir"].replace("\\", "/") + "/img"
184:     out_dir = cfg["output_dir"].replace("\\", "/")
185:     out_name = cfg["output_name"]
186:     lr = cfg["learning_rate"]
187:     dim = cfg["network_dim"]
188:     save_every = cfg.get("save_every_n_epochs", 250)
189:     res = cfg["resolution"]
190:     bs = cfg["batch_size"]
191:     seed = cfg["seed"]
192:     steps = int(cfg.get("steps") or 0) or int(cfg["epochs"]) * 100
193: 
194:     triggers = [t.strip() for t in cfg.get("trigger", "").split(",") if t.strip()]
195:     trigger = triggers[0] if triggers else ""
196: 
197:     # sample prompts: fluxgym-style textarea -> upstream sample: block (steps-based)
198:     prompts = cfg.get("sample_prompts") or []
199:     sample_every = int(cfg.get("sample_every") or 0)
200:     steps_per_epoch = max(1, steps // max(1, int(cfg["epochs"])))
201:     if prompts and sample_every > 0:
202:         yaml_lines = "\n".join(f'          - "{_yaml_str(p)}"' for p in prompts)
203:         sample_block = f"""      sample:
204:         sampler: "flowmatch" # must match train.noise_scheduler
205:         sample_every: {steps_per_epoch * sample_every}
206:         sample_start_step: 0
207:         width: {res}
208:         height: {res}
209:         prompts:
210: {yaml_lines}
211:         neg: ""
212:         seed: {seed}
213:         walk_seed: true
214:         guidance_scale: 3
215:         sample_steps: 25
216: """
217:     else:
218:         sample_block = ""
219: 
220:     return f"""job: extension
221: config:
222:   name: "{_yaml_str(out_name)}"
223:   process:
224:     - type: sd_trainer
225:       training_folder: "{_yaml_str(out_dir)}"
226:       device: cuda:0{f'\n      trigger_word: "{_yaml_str(trigger)}"' if trigger else ""}
227:       network:
228:         type: lora
229:         linear: {dim}
230:         linear_alpha: {dim}
231:       save:
232:         dtype: float16
233:         save_every: {save_every}
234:         max_step_saves_to_keep: 4
235:       datasets:
236:         - folder_path: "{_yaml_str(train_dir)}"
237:           caption_ext: "txt"
238:           caption_dropout_rate: 0.05
239:           shuffle_tokens: false
240:           cache_latents_to_disk: true
241:           resolution: [{res}, {res}]
242:       train:
243:         batch_size: {bs}
244:         steps: {steps}
245:         gradient_accumulation: 1
246:         # caching text embeddings is required for 24GB cards
247:         cache_text_embeddings: true
248:         train_unet: true
249:         train_text_encoder: false
250:         gradient_checkpointing: true
251:         noise_scheduler: "flowmatch"
252:         optimizer: "adamw8bit"
253:         lr: {lr}
254:         seed: {seed}
255:         dtype: bf16
256: {sample_block}      model:
257:         name_or_path: "{base_repo}"
258:         arch: "{arch}"
259:         quantize: true
260:         quantize_te: true
261:         qtype_te: "qfloat8"
262:         low_vram: true
263: """
264: 
265: 
266: def _atk_commands(cfg):
267:     """ai-toolkit path: embed the YAML in the script, run.py, report output."""
268:     py = cfg["python_exe"]
269:     atk = cfg["backend_dir"]
270:     yaml_path = os.path.join(cfg["output_dir"], cfg["output_name"] + ".atk.yml")
271:     out_dir = cfg["output_dir"]
272:     out_name = cfg["output_name"]
273: 
274:     yaml_text = _atk_yaml(cfg)
275:     # single-quoted here-string: PowerShell does no interpolation, YAML stays verbatim
276:     write = (
277:         "@'\n"
278:         f"{yaml_text}"
279:         "\n'@ | Set-Content -Encoding UTF8 \"" + yaml_path.replace('"', '`"') + "\"\n"
280:     )
281:     run = _ps([f"{py}", "run.py", _ps_quote(yaml_path)])
282:     body = (
283:         f'Write-Host "=== training Qwen-Image-2.1 LoRA (ai-toolkit) ==="\n'
284:         f"{write}\n"
285:         f"{run}\n"
286:         f'if ($LASTEXITCODE -ne 0) {{ throw "training failed" }}\n\n'
287:         f'Write-Host "DONE: {out_dir}\\{out_name}.safetensors"\n'
288:     )
289:     return body, "ai-toolkit"
290: 
291: 
292: def build_commands(cfg):
293:     """Route to the right backend and return (script_body, backend_label)."""
294:     if cfg["arch"] in MUSUBI_SCRIPTS:
295:         return _musubi_commands(cfg)
296:     if cfg["arch"] == "qwen_image_2":
297:         return _atk_commands(cfg)
298:     raise RuntimeError(f"Unknown architecture: {cfg['arch']}")
299: 
300: 
301: def full_script(cfg):
302:     body, backend = build_commands(cfg)
303:     header = (
304:         f"# fluxgym-pro :: {ARCH_LABELS[cfg['arch']]} LoRA :: {cfg['output_name']}\n"
305:         f"# backend: {backend}\n"
306:         f"# generated by fluxgym-pro — edit values in the UI, not here\n"
307:         f"$ErrorActionPreference = 'Stop'\n"
308:         f"Set-Location {_ps_quote(cfg['backend_dir'])}\n\n"
309:     )
310:     return header + body
