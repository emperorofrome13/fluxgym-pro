2: """
3: fluxgym-pro command generator.
4: 
5: Pure-Python, no torch / no GPU: builds the exact training command sequence for
6: each architecture.
7: 
8: Backends (verified against upstream docs, 2026-02):
9:   - Z-Image        -> musubi-tuner  (zimage_cache_latents.py, zimage_train_network.py)
10:   - Krea 2         -> musubi-tuner  (krea2_cache_latents.py,   krea2_train_network.py)
11:   - Qwen-Image-2.1 -> ai-toolkit    (run.py + generated YAML; musubi-tuner has no
12:                                       Qwen-Image-2.1 support yet, only the older
13:                                       20B Qwen-Image)
14: """
15: 
16: import os
17: 
18: # arch -> musubi-tuner script stems. qwen_image_2 is handled by ai-toolkit.
19: MUSUBI_SCRIPTS = {
20:     "zimage": {
21:         "cache_latents": "zimage_cache_latents.py",
22:         "cache_text": "zimage_cache_text_encoder_outputs.py",
23:         "train": "zimage_train_network.py",
24:     },
25:     "krea2": {
26:         "cache_latents": "krea2_cache_latents.py",
27:         "cache_text": "krea2_cache_text_encoder_outputs.py",
28:         "train": "krea2_train_network.py",
29:     },
30: }
31: 
32: # arch -> network module inside musubi-tuner's networks/ package
33: NETWORK_MODULE = {"zimage": "networks.lora_zimage", "krea2": "networks.lora_krea2"}
34: 
35: # flow-matching sampling settings per musubi arch (straight from upstream docs)
36: FLOW_ARGS = {
37:     "zimage": "--timestep_sampling shift --weighting_scheme none --discrete_flow_shift 2.0",
38:     "krea2": "--timestep_sampling shift --weighting_scheme none --discrete_flow_shift 2.5",
39: }
40: 
41: ARCH_LABELS = {
42:     "zimage": "Z-Image",
43:     "krea2": "Krea 2",
44:     "qwen_image_2": "Qwen-Image-2.1",
45: }

48: def _ps(lines):
49:     """Join command lines with PowerShell continuation backticks (no dangling)."""
50:     return " `\n  ".join(lines)
51: 
52: 
53: def _ps_quote(s):
54:     """Quote a value as a single PowerShell argument.
55: 
56:     Doubles backticks and escapes double quotes so a value from the UI can
57:     never terminate the quoted argument or escape into code. CR/LF are
58:     neutralized: inside a double-quoted string a raw newline would break the
59:     argument (and the line) apart.
60:     """
61:     return ('"' + str(s).replace("\r\n", " ").replace("\r", " ")
62:             .replace("\n", " ").replace("`", "``").replace('"', '`"') + '"')
63: 
64: 
65: def _yaml_str(s):
66:     """Escape a value for a YAML double-quoted scalar (no newlines, no breaks)."""
67:     return (str(s).replace("\r", " ").replace("\n", " ")
68:             .replace("\\", "\\\\").replace('"', '\\"'))

71: def dataset_toml(train_dir, resolution, batch_size, repeats=1):
72:     """musubi-tuner dataset config (docs/dataset_config.md schema)."""
73:     r = f"[{resolution}, {resolution}]"
74:     return (
75:         "# generated by fluxgym-pro\n"
76:         "[general]\n"
77:         f"resolution = {r}\n"
78:         "caption_extension = \".txt\"\n"
79:         f"batch_size = {batch_size}\n"
80:         "\n"
81:         "[[datasets]]\n"
82:         f"image_directory = \"{os.path.abspath(os.path.join(train_dir, 'img')).replace(os.sep, '/')}\"\n"
83:         f"num_repeats = {repeats}\n"
84:     )

87: def _musubi_commands(cfg):
88:     """musubi-tuner path: cache latents -> cache text encoder outputs -> train.
89: 
90:     Scripts run from the musubi-tuner repo root as `python src/musubi_tuner/<script>`.
91:     """
92:     arch = cfg["arch"]
93:     scripts = MUSUBI_SCRIPTS[arch]
94:     toml = cfg["toml_path"]
95:     py = cfg["python_exe"]
96:     out_dir = cfg["output_dir"]
97:     out_name = cfg["output_name"]
98:     epochs = cfg["epochs"]
99:     lr = cfg["learning_rate"]
100:     dim = cfg["network_dim"]
101:     alpha = int(cfg.get("network_alpha") or dim)
102:     seed = cfg["seed"]
103: 
104:     vae = cfg.get("vae_path")
105:     te = cfg.get("text_encoder_path")
106:     dit = cfg.get("dit_path")
107:     save_every = cfg.get("save_every_n_epochs", 1)
108:     prompts = cfg.get("sample_prompts") or []
109:     sample_every = int(cfg.get("sample_every") or 0)
110:     sampling = bool(prompts) and sample_every > 0
111:     prompts_path = ""
112:     if sampling:
113:         # command_gen may be called directly (tests, previews); keep its own
114:         # file writes safe even if the caller skipped build_cfg validation.
115:         fs_name = "".join(c if c not in '\\/:*?"<>|' else "_" for c in out_name)
116:         prompts_path = os.path.join(cfg["output_dir"],
117:                                     f"{fs_name}_sample_prompts.txt")
118:         with open(prompts_path, "w", encoding="utf-8") as f:
119:             for p in prompts:
120:                 f.write(p.replace("\n", " ").strip() + "\n")
121: 
122:     if not (vae and os.path.exists(vae)):
123:         raise RuntimeError(f"VAE weights not found: {vae}")
124:     if not (te and os.path.exists(te)):
125:         raise RuntimeError(f"Text encoder weights not found: {te}")
126:     if not (dit and os.path.exists(dit)):
127:         raise RuntimeError(f"DiT weights not found: {dit}")
128: 
129:     cache_lat = _ps([f"{py}", "src/musubi_tuner/" + scripts["cache_latents"],
130:                      "--dataset_config", _ps_quote(toml), "--vae", _ps_quote(vae)])
131:     cache_te = _ps([f"{py}", "src/musubi_tuner/" + scripts["cache_text"],
132:                     "--dataset_config", _ps_quote(toml), "--text_encoder", _ps_quote(te),
133:                     "--batch_size", "16"])
134:     train = _ps([
135:         "accelerate launch --num_cpu_threads_per_process 1 --mixed_precision bf16",
136:         "src/musubi_tuner/" + scripts["train"],
137:         "--dit", _ps_quote(dit),
138:         "--vae", _ps_quote(vae),
139:         # krea2 only needs the TE during training if sample images are generated
140:         *(["--text_encoder", _ps_quote(te)] if (arch != "krea2" or sampling) else []),
141:         "--dataset_config", _ps_quote(toml),
142:         "--sdpa --mixed_precision bf16",
143:         *(["--fp8_base --fp8_scaled"] if cfg.get("fp8_base") else []),
144:         FLOW_ARGS[arch],
145:         "--optimizer_type adamw8bit",
146:         "--learning_rate", str(lr),
147:         "--gradient_checkpointing",
148:         "--max_data_loader_n_workers 2 --persistent_data_loader_workers",
149:         "--network_module", NETWORK_MODULE[arch],
150:         "--network_dim", str(dim),
151:         "--network_alpha", str(alpha),
152:         "--max_train_epochs", str(epochs),
153:         f"--save_every_n_epochs {save_every} --save_state",
154:         *(["--sample_prompts", _ps_quote(prompts_path), "--sample_every_n_epochs", str(sample_every)]
155:           if sampling else []),
156:         "--seed", str(seed),
157:         "--output_dir", _ps_quote(out_dir),
158:         "--output_name", _ps_quote(out_name),
159:     ])
160: 
161:     body = (
162:         f'Write-Host "=== [1/3] caching latents ({ARCH_LABELS[arch]}) ==="\n'
163:         f"{cache_lat}\n"
164:         f'if ($LASTEXITCODE -ne 0) {{ throw "latent caching failed" }}\n\n'
165:         f'Write-Host "=== [2/3] caching text encoder outputs ==="\n'
166:         f"{cache_te}\n"
167:         f'if ($LASTEXITCODE -ne 0) {{ throw "text encoder caching failed" }}\n\n'
168:         f'Write-Host "=== [3/3] training LoRA ==="\n'
169:         f"{train}\n"
170:         f'if ($LASTEXITCODE -ne 0) {{ throw "training failed" }}\n\n'
171:         f'Write-Host "DONE: {out_dir}\\{out_name}.safetensors"\n'
172:     )
173:     return body, "musubi-tuner"

176: def _atk_yaml(cfg):
177:     """ai-toolkit training config for Qwen-Image-2.1 (day-0 upstream support).
178: 
179:     Schema mirrors ai-toolkit's sd_trainer examples: job=extension, one
180:     sd_trainer process, model.arch selects the architecture implementation.
181:     """
182:     arch = "qwen_image_2"
183:     base_repo = cfg["repo"]
184:     train_dir = cfg["train_data_dir"].replace("\\", "/") + "/img"
185:     out_dir = cfg["output_dir"].replace("\\", "/")
186:     out_name = cfg["output_name"]
187:     lr = cfg["learning_rate"]
188:     dim = cfg["network_dim"]
189:     save_every = cfg.get("save_every_n_epochs", 250)
190:     res = cfg["resolution"]
191:     bs = cfg["batch_size"]
192:     seed = cfg["seed"]
193:     steps = int(cfg.get("steps") or 0) or int(cfg["epochs"]) * 100
194: 
195:     triggers = [t.strip() for t in cfg.get("trigger", "").split(",") if t.strip()]
196:     trigger = triggers[0] if triggers else ""
197: 
198:     # sample prompts: fluxgym-style textarea -> upstream sample: block (steps-based)
199:     prompts = cfg.get("sample_prompts") or []
200:     sample_every = int(cfg.get("sample_every") or 0)
201:     steps_per_epoch = max(1, steps // max(1, int(cfg["epochs"])))
202:     if prompts and sample_every > 0:
203:         yaml_lines = "\n".join(f'          - "{_yaml_str(p)}"' for p in prompts)
204:         sample_block = f"""      sample:
205:         sampler: "flowmatch" # must match train.noise_scheduler
206:         sample_every: {steps_per_epoch * sample_every}
207:         sample_start_step: 0
208:         width: {res}
209:         height: {res}
210:         prompts:
211: {yaml_lines}
212:         neg: ""
213:         seed: {seed}
214:         walk_seed: true
215:         guidance_scale: 3
216:         sample_steps: 25
217: """
218:     else:
219:         sample_block = ""
220: 
221:     return f"""job: extension
222: config:
223:   name: "{_yaml_str(out_name)}"
224:   process:
225:     - type: sd_trainer
226:       training_folder: "{_yaml_str(out_dir)}"
227:       device: cuda:0{f'\n      trigger_word: "{_yaml_str(trigger)}"' if trigger else ""}
228:       network:
229:         type: lora
230:         linear: {dim}
231:         linear_alpha: {dim}
232:       save:
233:         dtype: float16
234:         save_every: {save_every}
235:         max_step_saves_to_keep: 4
236:       datasets:
237:         - folder_path: "{_yaml_str(train_dir)}"
238:           caption_ext: "txt"
239:           caption_dropout_rate: 0.05
240:           shuffle_tokens: false
241:           cache_latents_to_disk: true
242:           resolution: [{res}, {res}]
243:       train:
244:         batch_size: {bs}
245:         steps: {steps}
246:         gradient_accumulation: 1
247:         # caching text embeddings is required for 24GB cards
248:         cache_text_embeddings: true
249:         train_unet: true
250:         train_text_encoder: false
251:         gradient_checkpointing: true
252:         noise_scheduler: "flowmatch"
253:         optimizer: "adamw8bit"
254:         lr: {lr}
255:         seed: {seed}
256:         dtype: bf16
257: {sample_block}      model:
258:         name_or_path: "{base_repo}"
259:         arch: "{arch}"
260:         quantize: true
261:         quantize_te: true
262:         qtype_te: "qfloat8"
263:         low_vram: true
264: """

267: def _atk_commands(cfg):
268:     """ai-toolkit path: embed the YAML in the script, run.py, report output."""
269:     py = cfg["python_exe"]
270:     atk = cfg["backend_dir"]
271:     yaml_path = os.path.join(cfg["output_dir"], cfg["output_name"] + ".atk.yml")
272:     out_dir = cfg["output_dir"]
273:     out_name = cfg["output_name"]
274: 
275:     yaml_text = _atk_yaml(cfg)
276:     # single-quoted here-string: PowerShell does no interpolation, YAML stays verbatim
277:     write = (
278:         "@'\n"
279:         f"{yaml_text}"
280:         "\n'@ | Set-Content -Encoding UTF8 \"" + yaml_path.replace('"', '`"') + "\"\n"
281:     )
282:     run = _ps([f"{py}", "run.py", _ps_quote(yaml_path)])
283:     body = (
284:         f'Write-Host "=== training Qwen-Image-2.1 LoRA (ai-toolkit) ==="\n'
285:         f"{write}\n"
286:         f"{run}\n"
287:         f'if ($LASTEXITCODE -ne 0) {{ throw "training failed" }}\n\n'
288:         f'Write-Host "DONE: {out_dir}\\{out_name}.safetensors"\n'
289:     )
290:     return body, "ai-toolkit"

293: def build_commands(cfg):
294:     """Route to the right backend and return (script_body, backend_label)."""
295:     if cfg["arch"] in MUSUBI_SCRIPTS:
296:         return _musubi_commands(cfg)
297:     if cfg["arch"] == "qwen_image_2":
298:         return _atk_commands(cfg)
299:     raise RuntimeError(f"Unknown architecture: {cfg['arch']}")

301: 
302: def full_script(cfg):
303:     body, backend = build_commands(cfg)
304:     header = (
305:         f"# fluxgym-pro :: {ARCH_LABELS[cfg['arch']]} LoRA :: {cfg['output_name']}\n"
306:         f"# backend: {backend}\n"
307:         f"# generated by fluxgym-pro — edit values in the UI, not here\n"
308:         f"$ErrorActionPreference = 'Stop'\n"
309:         f"Set-Location {_ps_quote(cfg['backend_dir'])}\n\n"
310:     )
311:     return header + body
