61: def create_dataset(name, files, trigger):
62:     name = _safe_name(name, "dataset name")
63:     if not files:
64:         raise gr.Error("Add at least one image.")
65:     img_dir = os.path.join(DATASETS_DIR, name, "img")
66:     os.makedirs(img_dir, exist_ok=True)
67:     trigger = (trigger or "").strip()
68:     n = 0
69:     for src in files:
70:         path = src.name if hasattr(src, "name") else src  # gradio 4/5 compat
71:         base, _ = os.path.splitext(os.path.basename(path))
72:         dst = os.path.join(img_dir, os.path.basename(path))
73:         shutil.copy2(path, dst)
74:         cap = os.path.join(img_dir, base + ".txt")
75:         content = ""
76:         if os.path.exists(cap):
77:             with open(cap, "r", encoding="utf-8") as f:
78:                 content = f.read().strip()
79:         if trigger and not content.startswith(trigger):
80:             content = f"{trigger} {content}".strip()
81:         with open(cap, "w", encoding="utf-8") as f:
82:             f.write(content)
83:         n += 1
84:     with open(os.path.join(DATASETS_DIR, name, "trigger.txt"), "w",
85:               encoding="utf-8") as f:
86:         f.write(trigger)
87:     return (f"Dataset '{name}' created with {n} images.",
88:             gr.update(choices=dataset_names(), value=name))

102: def auto_caption(dataset_ui, dataset_dd):
103:     """Run AI captioning (WD14 tagger) over a dataset in the background, CPU-only."""
104:     name = _safe_name((dataset_ui or "").strip() or (dataset_dd or "").strip(),
105:                       "dataset name")
106:     if not name:
107:         raise gr.Error("Create a dataset first, then run AI captions.")
108:     img_dir = os.path.join(DATASETS_DIR, name, "img")
109:     if not os.path.isdir(img_dir):
110:         raise gr.Error(f"Dataset '{name}' not found. Create it first.")
111:     if not any(f.lower().endswith(IMG_EXTS) for f in os.listdir(img_dir)):
112:         raise gr.Error(f"Dataset '{name}' has no images.")
113: 
114:     cap_py = os.path.join(ROOT, "caption.py")
115:     ps1 = os.path.join(DATASETS_DIR, name, "caption.ps1")
116:     log_path = os.path.join(DATASETS_DIR, name, "caption.log")
117:     py = sys.executable
118:     with open(ps1, "w", encoding="utf-8-sig") as f:
119:         f.write(
120:             "$ErrorActionPreference = 'Stop'\n"
121:             f"$py = \"{py}\"\n"
122:             "& $py -c \"import onnxruntime, huggingface_hub, numpy, PIL\" 2>$null\n"
123:             "if ($LASTEXITCODE -ne 0) { & $py -m pip install --quiet onnxruntime huggingface_hub numpy pillow }\n"
124:             f"& $py \"{cap_py}\" \"{img_dir}\" \"{dataset_trigger(name)}\"\n"
125:             "if ($LASTEXITCODE -ne 0) { throw \"AI captioning failed - see caption.log\" }\n"
126:         )
127:     log = open(log_path, "w", encoding="utf-8", buffering=1)
128:     proc = subprocess.Popen(
129:         ["powershell", "-NoProfile", "-ExecutionPolicy", "Bypass", "-File", ps1],
130:         cwd=DATASETS_DIR, stdout=log, stderr=subprocess.STDOUT)
131:     CAPTION_JOBS[name] = proc
132:     return (f"AI captioning started for '{name}' in the background (CPU-only, "
133:             f"first run downloads the WD14 tagger model). "
134:             f"Existing captions are kept. Log: {log_path}")

159: def _resolve_weights(m, backend):
160:     """Weight paths resolve to models/<repo>/..., then the backend folder; absolute wins."""
161:     repo_dir = os.path.join(MODELS_DIR, (m.get("repo") or "").replace("/", "_"))
162:     out = {}
163:     for key in ("dit_path", "vae_path", "text_encoder_path"):
164:         val = (m.get(key) or "").strip()
165:         if not val:
166:             continue
167:         if os.path.isabs(val):
168:             out[key] = os.path.abspath(val)
169:             continue
170:         for base in (repo_dir, backend):
171:             cand = os.path.abspath(os.path.join(base, val))
172:             if os.path.exists(cand):
173:                 out[key] = cand
174:                 break
175:         else:
176:             out[key] = os.path.abspath(os.path.join(repo_dir, val))
177:     return out

190: def build_cfg(base_model, dataset_name, output_name, backend_dir,
191:               learning_rate, network_dim, network_alpha, resolution,
192:               epochs, batch_size, save_every_n_epochs, mixed_precision,
193:               sample_prompts, sample_every):
194:     if base_model not in MODELS:
195:         raise gr.Error("Pick a base model.")
196:     output_name = _safe_name(output_name, "LoRA name")
197:     train_dir = os.path.join(DATASETS_DIR, dataset_name or "")
198:     if _n_images(train_dir) == 0:
199:         raise gr.Error(f"Dataset '{dataset_name}' has no images. Create one first.")
200: 
201:     m = MODELS[base_model]
202:     arch = m["arch"]
203:     backend = os.path.abspath((backend_dir or "").strip() or m["backend_dir"])
204:     if not os.path.isdir(backend):
205:         raise gr.Error(f"Backend folder not found: {backend}\n"
206:                        f"Clone it there (see README) or fix the path.")
207: 
208:     out_dir = os.path.join(OUTPUTS_DIR, output_name)
209:     os.makedirs(out_dir, exist_ok=True)
210:     prompts = [l.strip() for l in (sample_prompts or "").splitlines() if l.strip()]
211: 
212:     cfg = {
213:         "arch": arch,
214:         "backend_dir": backend,
215:         "python_exe": _pick_python(backend),
216:         "train_data_dir": train_dir,
217:         "toml_path": os.path.join(out_dir, "dataset.toml"),
218:         "output_dir": out_dir,
219:         "output_name": output_name,
220:         "repo": m.get("repo", ""),
221:         "trigger": dataset_trigger(dataset_name),
222:         "resolution": int(resolution),
223:         "batch_size": int(batch_size),
224:         "epochs": int(epochs),
225:         "save_every_n_epochs": int(save_every_n_epochs),
226:         "learning_rate": float(learning_rate),
227:         "network_dim": int(network_dim),
228:         "network_alpha": int(network_alpha),
229:         "mixed_precision": mixed_precision or "bf16",
230:         "fp8_base": bool(m.get("fp8_base", False)),
231:         "repeats": int(m.get("num_repeats", 1)),
232:         "seed": int(m.get("seed", 0)),
233:         "sample_prompts": prompts,
234:         "sample_every": int(sample_every or 0),
235:     }
236:     cfg.update(_resolve_weights(m, backend))
237: 
238:     if arch == "qwen_image_2":
239:         n = _n_images(train_dir)
240:         bs = max(1, cfg["batch_size"])
241:         cfg["steps"] = max(1, (n * cfg["repeats"] * cfg["epochs"]) // bs)
242:     return cfg
